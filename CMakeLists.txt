cmake_minimum_required(VERSION 3.22)
project(UltimateHighGainDSP VERSION 0.1.0 LANGUAGES C CXX)
enable_testing()
foreach(t test presence_test tube_test quick_test preamp_test jose_test extra_test architecture_test shared_test stock_test evh_test)
 if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${t}.cpp")
  add_executable(uhg_${t} ${t}.cpp)
  target_compile_features(uhg_${t} PRIVATE cxx_std_17)
  add_test(NAME ${t} COMMAND uhg_${t})
 endif()
endforeach()
option(UHG_BUILD_PLUGIN "Build JUCE VST3" OFF)
if(UHG_BUILD_PLUGIN)
 include(FetchContent)
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
 FetchContent_MakeAvailable(JUCE)
 juce_add_plugin(UltimateHighGain COMPANY_NAME "Jimijaymes" PLUGIN_MANUFACTURER_CODE Jjms PLUGIN_CODE Uhg1 FORMATS VST3 PRODUCT_NAME "Ultimate High Gain" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
 target_sources(UltimateHighGain PRIVATE Plugin.cpp)
 target_compile_features(UltimateHighGain PRIVATE cxx_std_17)
 target_compile_definitions(UltimateHighGain PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
 target_link_libraries(UltimateHighGain PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_recommended_config_flags)
endif()
